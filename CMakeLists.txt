cmake_minimum_required(VERSION 3.21)
project(NatureLoader VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(LibArchive REQUIRED)
qt_add_resources(ASSET_SOURCES resources/assets.qrc)
add_library(nature_core src/core.cpp src/rom_catalog.cpp src/texture_decoder.cpp src/native_mods.cpp ${ASSET_SOURCES})
target_include_directories(nature_core PUBLIC src)
target_link_libraries(nature_core PUBLIC Qt6::Widgets LibArchive::LibArchive)
add_executable(natureloader WIN32 src/main.cpp src/rom_panel.cpp)
target_link_libraries(natureloader PRIVATE nature_core)
set(NATURE_NATIVE_RUNTIME_DIRECTORY "" CACHE PATH "Optional local native research runtime checkout (not bundled)")
if(NATURE_NATIVE_RUNTIME_DIRECTORY)
  target_compile_definitions(natureloader PRIVATE NATURE_NATIVE_RUNTIME_DIRECTORY="${NATURE_NATIVE_RUNTIME_DIRECTORY}")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(nature_tests tests/core_test.cpp)
  target_link_libraries(nature_tests PRIVATE nature_core)
  add_test(NAME core COMMAND nature_tests)
  add_executable(rom_catalog_tests tests/rom_catalog_test.cpp)
  target_link_libraries(rom_catalog_tests PRIVATE nature_core)
  add_test(NAME rom_catalog COMMAND rom_catalog_tests)
  add_executable(texture_decoder_tests tests/texture_decoder_test.cpp)
  target_link_libraries(texture_decoder_tests PRIVATE nature_core)
  add_test(NAME texture_decoder COMMAND texture_decoder_tests)
  add_test(NAME launcher_smoke COMMAND natureloader --smoke-test)
  set_tests_properties(launcher_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set(NATURE_TEST_ROM "" CACHE FILEPATH "Optional local ROM for integration test; never bundled")
  if(NATURE_TEST_ROM)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    add_test(NAME rom_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:natureloader> ${NATURE_TEST_ROM})
  endif()
endif()
install(TARGETS natureloader RUNTIME DESTINATION bin)
install(FILES THIRD_PARTY_NOTICES.md DESTINATION share/natureloader)
install(DIRECTORY licenses/ DESTINATION share/natureloader/licenses)
