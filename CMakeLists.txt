cmake_minimum_required(VERSION 3.21)
project(NatureLoader VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(CTest)
include(GNUInstallDirs)
add_library(nature_modcore STATIC
  modcore/json.cpp modcore/util.cpp modcore/version.cpp modcore/manifest.cpp
  modcore/resolver.cpp modcore/registry.cpp modcore/vfs.cpp modcore/hooks.cpp
  modcore/save.cpp modcore/plugin.cpp modcore/session.cpp modcore/capi.cpp)
target_include_directories(nature_modcore PUBLIC modcore)
target_link_libraries(nature_modcore PUBLIC ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(nature_modcore PRIVATE Advapi32)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/modcore/data/base_catalog.json" NATURE_BASE_CATALOG)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS modcore/data/base_catalog.json)
configure_file(modcore/data/catalog.h.in generated/modcore_catalog.h @ONLY)
add_executable(naturemod tools/naturemod.cpp)
target_include_directories(naturemod PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(naturemod PRIVATE nature_modcore)
install(TARGETS naturemod RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(BUILD_TESTING)
  add_library(plugin_sample SHARED tests/modcore/plugin_sample.c)
  target_include_directories(plugin_sample PRIVATE modcore)
  add_executable(modcore_tests tests/modcore/modcore_tests.cpp)
  target_link_libraries(modcore_tests PRIVATE nature_modcore)
  target_compile_definitions(modcore_tests PRIVATE PLUGIN_SO="$<TARGET_FILE:plugin_sample>")
  add_dependencies(modcore_tests plugin_sample)
  add_test(NAME modcore COMMAND modcore_tests)
  add_library(plugin_fail_fixture SHARED tests/modcore/plugin_lifetime_tests.cpp)
  target_include_directories(plugin_fail_fixture PRIVATE modcore)
  target_compile_definitions(plugin_fail_fixture PRIVATE NATURE_PLUGIN_FAIL_FIXTURE)
  add_executable(modcore_plugin_lifetime_tests tests/modcore/plugin_lifetime_tests.cpp)
  target_link_libraries(modcore_plugin_lifetime_tests PRIVATE nature_modcore)
  target_compile_definitions(modcore_plugin_lifetime_tests PRIVATE
    PLUGIN_SO="$<TARGET_FILE:plugin_sample>" FAIL_PLUGIN_SO="$<TARGET_FILE:plugin_fail_fixture>")
  add_dependencies(modcore_plugin_lifetime_tests plugin_sample plugin_fail_fixture)
  add_test(NAME modcore_plugin_lifetime COMMAND modcore_plugin_lifetime_tests)
  add_executable(modcore_boundary_tests tests/modcore/boundary_tests.cpp)
  target_link_libraries(modcore_boundary_tests PRIVATE nature_modcore)
  add_test(NAME modcore_boundaries COMMAND modcore_boundary_tests)
  add_executable(modcore_script_boundary_tests tests/modcore/script_boundary_tests.cpp)
  target_link_libraries(modcore_script_boundary_tests PRIVATE nature_modcore)
  add_test(NAME modcore_script_boundaries COMMAND modcore_script_boundary_tests)
  add_executable(fake_core tests/modcore/fake_core.cpp)
  target_link_libraries(fake_core PRIVATE nature_modcore)
  add_test(NAME modcore_end_to_end COMMAND fake_core examples/v2)
  add_test(NAME modcore_examples_load COMMAND naturemod check examples/v2)
  add_test(NAME modcore_cli_errors COMMAND ${CMAKE_COMMAND}
    "-DNATUREMOD=$<TARGET_FILE:naturemod>"
    "-DEXAMPLES=${CMAKE_CURRENT_SOURCE_DIR}/examples/v2"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/modcore/cli_errors.cmake")
  set_tests_properties(modcore modcore_end_to_end modcore_examples_load
    PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
option(NATURE_BUILD_LAUNCHER "Build the Qt desktop launcher" ON)
if(NATURE_BUILD_LAUNCHER)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(LibArchive REQUIRED)
qt_add_resources(ASSET_SOURCES resources/assets.qrc)
add_library(nature_core src/core.cpp src/rom_catalog.cpp src/texture_decoder.cpp src/native_mods.cpp src/modcore_library.cpp ${ASSET_SOURCES})
target_include_directories(nature_core PUBLIC src)
target_include_directories(nature_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(nature_core PUBLIC Qt6::Widgets LibArchive::LibArchive PRIVATE nature_modcore)
add_executable(natureloader WIN32 src/main.cpp src/rom_panel.cpp src/modcore_panel.cpp)
target_link_libraries(natureloader PRIVATE nature_core)
set(NATURE_NATIVE_RUNTIME_DIRECTORY "" CACHE PATH "Optional local native research runtime checkout (not bundled)")
if(NATURE_NATIVE_RUNTIME_DIRECTORY)
  target_compile_definitions(natureloader PRIVATE NATURE_NATIVE_RUNTIME_DIRECTORY="${NATURE_NATIVE_RUNTIME_DIRECTORY}")
endif()
if(BUILD_TESTING)
  add_executable(modcore_library_tests tests/modcore_library_test.cpp)
  target_link_libraries(modcore_library_tests PRIVATE nature_core)
  add_test(NAME modcore_library COMMAND modcore_library_tests)
  add_executable(nature_tests tests/core_test.cpp)
  target_link_libraries(nature_tests PRIVATE nature_core)
  add_test(NAME core COMMAND nature_tests)
  add_executable(rom_catalog_tests tests/rom_catalog_test.cpp)
  target_link_libraries(rom_catalog_tests PRIVATE nature_core)
  add_test(NAME rom_catalog COMMAND rom_catalog_tests)
  add_executable(texture_decoder_tests tests/texture_decoder_test.cpp)
  target_link_libraries(texture_decoder_tests PRIVATE nature_core)
  add_test(NAME texture_decoder COMMAND texture_decoder_tests)
  add_test(NAME launcher_smoke COMMAND natureloader --smoke-test)
  set_tests_properties(launcher_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set(NATURE_TEST_ROM "" CACHE FILEPATH "Optional local ROM for integration test; never bundled")
  if(NATURE_TEST_ROM)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    add_test(NAME rom_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:natureloader> ${NATURE_TEST_ROM})
  endif()
endif()
install(TARGETS natureloader RUNTIME DESTINATION bin)
endif()
install(FILES THIRD_PARTY_NOTICES.md DESTINATION share/natureloader)
install(DIRECTORY licenses/ DESTINATION share/natureloader/licenses)
